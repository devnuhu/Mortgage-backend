 565:         private async Task<List<WorkloadDistributionDto>> MapToWorkloadDtosAsync(List<ReviewerPerformanceMetrics> metrics)
 566:         {
 567:             var totalAssignments = metrics.Sum(m => m.TotalAssignments);
 568:             var maxCapacityHours = 160.0; // Standard monthly capacity (40 hrs/week * 4 weeks)
 569: 
 570:             var now = DateTime.UtcNow;
 571:             var oneWeekAgo = now.AddDays(-7);
 572:             var twoWeeksAgo = now.AddDays(-14);
 573: 
 574:             // OPTIMIZATION: Batch load all assignments for all reviewers in ONE query
 575:             // Using a simpler query pattern to avoid WITH clause issues
 576:             var allAssignments = await _context.ApplicationAssignments
 577:                 .AsNoTracking()
 578:                 .Include(a => a.Application)
 579:                 .ToListAsync();
 580: 
 581:             // Group assignments by reviewer for fast lookup
 582:             var assignmentsByReviewer = allAssignments
 583:                 .GroupBy(a => a.ReviewerId)
 584:                 .ToDictionary(g => g.Key, g => g.ToList());
 585: 
 586:             var result = new List<WorkloadDistributionDto>();
 587: 
 588:             foreach (var m in metrics)
 589:             {
 590:                 // Calculate quality metrics
 591:                 var totalDocumentsReviewed = m.DocumentsApproved + m.DocumentsRejected + m.DocumentsRevisionRequested;
 592:                 var escalationRate = m.TotalAssignments > 0 ? (m.EscalatedAssignments * 100.0 / m.TotalAssignments) : 0;
 593:                 var revisionRate = totalDocumentsReviewed > 0 ? (m.DocumentsRevisionRequested * 100.0 / totalDocumentsReviewed) : 0;
 594: 
 595:                 // Estimate rework percentage (applications that had rework)
 596:                 var reworkPercentage = m.TotalAssignments > 0 
 597:                     ? Math.Min(100, escalationRate + revisionRate / 2) 
 598:                     : 0;
 599: 
 600:                 // Calculate capacity metrics
 601:                 var capacityUtilization = (m.TotalHoursSpent / maxCapacityHours) * 100;
 602:                 var estimatedHoursRemaining = m.ActiveAssignments > 0 
 603:                     ? m.AverageHoursPerAssignment * m.ActiveAssignments 
 604:                     : 0;
 605: 
 606:                 var estimatedCompletionDate = estimatedHoursRemaining > 0
 607:                     ? DateTime.UtcNow.AddHours(estimatedHoursRemaining)
 608:                     : (DateTime?)null;
 609: 
 610:                 // Determine capacity status based on utilization
 611:                 var capacityStatus = capacityUtilization > 100 ? "Red" :
 612:                                     capacityUtilization > 85 ? "Yellow" : "Green";
 613: 
 614:                 // Calculate burnout risk
 615:                 var burnoutRisk = capacityUtilization > 100 ? "High" :
 616:                                  capacityUtilization > 80 ? "Medium" : "Low";
 617: 
 618:                 // OPTIMIZATION: Calculate SLA from pre-loaded in-memory data instead of database query
 619:                 var reviewerAssignments = assignmentsByReviewer.TryGetValue(m.ReviewerId, out var assignments) 
 620:                     ? assignments 
 621:                     : new List<ApplicationAssignment>();
 622: 
 623:                 var slaCompliance = CalculateSLAComplianceInMemory(reviewerAssignments);
 624: 
 625:                 // OPTIMIZATION: Calculate trend from pre-loaded in-memory data instead of database query
 626:                 var (weeksAvgHours, lastWeekHours, prevWeekHours) = CalculateWeeklyHoursTrendInMemory(
 627:                     reviewerAssignments, 
 628:                     now, oneWeekAgo, twoWeeksAgo);
 629: 
 630:                 var velocityChange = prevWeekHours > 0 ? ((lastWeekHours - prevWeekHours) / prevWeekHours) * 100 : 0;
 631: 
 632:                 var avgDocsPerApp = m.TotalAssignments > 0 ? (double)totalDocumentsReviewed / m.TotalAssignments : 0;
 633: 
 634:                 var workloadTrend = velocityChange > 5 ? "Increasing" :
 635:                                    velocityChange < -5 ? "Decreasing" : "Stable";
 636: 
 637:                 result.Add(new WorkloadDistributionDto
 638:                 {
 639:                     // Basic Info
 640:                     ReviewerId = m.ReviewerId,
 641:                     ReviewerName = m.Reviewer != null ? $"{m.Reviewer.FirstName} {m.Reviewer.LastName}" : "Unknown",
 642:                     ReviewerEmail = m.Reviewer?.Email ?? string.Empty,
 643: 
 644:                     // Workload Breakdown
 645:                     ActiveAssignments = m.ActiveAssignments,
 646:                     PendingAssignments = m.TotalAssignments - m.CompletedAssignments - m.ActiveAssignments,
 647:                     CompletedAssignments = m.CompletedAssignments,
 648:                     TotalAssignments = m.TotalAssignments,
 649:                     WorkloadPercentage = totalAssignments > 0 ? Math.Round((m.TotalAssignments * 100.0 / totalAssignments), 2) : 0,
 650: 
 651:                     // Performance Metrics
 652:                     ApprovalRatePercentage = Math.Round(m.ApprovalRatePercentage, 2),
 653:                     RejectionRatePercentage = Math.Round(m.RejectionRatePercentage, 2),
 654:                     RevisionRequestRatePercentage = Math.Round(m.RevisionRequestRatePercentage, 2),
 655:                     AverageTurnaroundTimeHours = Math.Round(m.AverageTurnaroundTimeHours, 2),
 656:                     PerformanceRating = m.PerformanceRating,
 657: 
 658:                     // Quality & Escalation
 659:                     EscalatedAssignments = m.EscalatedAssignments,
 660:                     EscalationRatePercentage = Math.Round(escalationRate, 2),
 661:                     RevisionRequestRate = Math.Round(revisionRate, 2),
 662:                     ReworkPercentage = Math.Round(reworkPercentage, 2),
 663:                     DocumentsPending = 0,
 664:                     DocumentsApproved = m.DocumentsApproved,
 665:                     DocumentsRejected = m.DocumentsRejected,
 666:                     DocumentsRevisionRequested = m.DocumentsRevisionRequested,
 667:                     AverageDocumentsPerApplication = Math.Round(avgDocsPerApp, 2),
 668: 
 669:                     // Hours Tracking
 670:                     TotalHoursSpent = Math.Round(m.TotalHoursSpent, 2),
 671:                     AverageHoursPerAssignment = Math.Round(m.AverageHoursPerAssignment, 2),
 672:                     EstimatedHoursRemaining = Math.Round(estimatedHoursRemaining, 2),
 673:                     EstimatedCompletionDate = estimatedCompletionDate,
 674: 
 675:                     // Capacity & Health
 676:                     MaxCapacityHours = maxCapacityHours,
 677:                     CapacityUtilizationPercentage = Math.Round(capacityUtilization, 2),
 678:                     AvailableCapacityHours = Math.Round(Math.Max(0, maxCapacityHours - m.TotalHoursSpent), 2),
 679:                     CapacityStatus = capacityStatus,
 680:                     BurnoutRisk = burnoutRisk,
 681:                     SLAComplianceRate = Math.Round(slaCompliance, 2),
 682: 
 683:                     // Trend & Velocity
 684:                     WeeklyWorkloadAverage = Math.Round(weeksAvgHours, 2),
 685:                     WorkloadTrend = workloadTrend,
 686:                     AverageHoursLastWeek = Math.Round(lastWeekHours, 2),
 687:                     AverageHoursPreviousWeek = Math.Round(prevWeekHours, 2),
 688:                     VelocityChangePercentage = Math.Round(velocityChange, 2),
 689:                     LastMetricsUpdateTime = m.LastCalculatedAt
 690:                 });
 691:             }
 692:

 694:         }

 696:         /// <summary>
 697:         /// OPTIMIZED: Calculate SLA compliance from pre-loaded in-memory data instead of database query
 698:         /// This replaces CalculateSLAComplianceAsync for performance
 699:         /// </summary>
 700:         private double CalculateSLAComplianceInMemory(List<ApplicationAssignment> assignments)
 701:         {
 702:             const int slaDaysTarget = 2; // 48-hour SLA
 703: 
 704:             var completedAssignments = assignments
 705:                 .Where(a => !a.IsActive && a.Application != null && a.Application.CompletedAt.HasValue)
 706:                 .ToList();
 707: 
 708:             if (!completedAssignments.Any())
 709:                 return 100.0; // No completed assignments = perfect compliance
 710: 
 711:             var compliantCount = completedAssignments.Count(a =>
 712:                 (a.Application.CompletedAt.Value - a.AssignedAt).TotalDays <= slaDaysTarget);
 713: 
 714:             return (compliantCount * 100.0 / completedAssignments.Count);
 715:         }
